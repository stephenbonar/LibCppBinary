338 void StoreLittleEndian(IntType value)
339 {
340 // Start the shift amount at 0 to obtain the least significant
341 // byte of value first, which is what we want for little endian.
342 int shiftAmount = 0;
343
344 for (int i = 0; i < IntSize; ++i)
345 {
346 // Get the next byte from value by shifting it to the least
347 // significant byte. By doing bitwise with the lsb mask, all
348 // other bytes are zeroed out.
349 rawData[i] =
350 static_cast<char>((value >> shiftAmount) & byteMask);
351
352 // Increase the shift amount to select the next byte.
353 shiftAmount += bitsPerByte;
354 }
355 }
356
360 void StoreBigEndian(IntType value)
361 {
362 // Because the byte order is reversed for big endian values, we
363 // start the shift amount towards the size and decrease to 0.
364 int shiftAmount = (IntSize - 1) * bitsPerByte;
365
366 for (int i = 0; i < IntSize; ++i)
367 {
368 // Get the next byte from value by shifting it to the least
369 // significant byte. By doing bitwise with the lsb mask, all
370 // other bytes are zeroed out.
371 rawData[i] =
372 static_cast<char>((value >> shiftAmount) & byteMask);
373
374 // Decrease the shift amount to select the next byte.
375 shiftAmount -= bitsPerByte;
376 }
377 }

381 IntType RetrieveLittleEndian() const
382 {
383 bool isNegative = false;
384
385 // Assemble the retrieved value in a 64-bit variable so it can hold
386 // all possible sized values. We make it unsigned so we don't have
387 // acccidental sign changes during the process.
388 uint64_t retrievedValue{ 0 };
389
390 // Start the shift amount at 0 to obtain the least significant
391 // byte of value first, which is what we want for little endian.
392 int shiftAmount = 0;
393
394 for (int i = 0; i < IntSize; ++i)
395 {
396 // First convert the raw byte char to an uint8_t so it can be
397 // bitwised OR'd to the rest of the retrieved value. We keep
398 // everything unsigned to avoid sign conversion errors.
399 uint8_t byte = static_cast<uint8_t>(rawData[i]);
400
401 // Determine if the most significant byte has the sign bit
402 // enabled to determine if the entire number is negative.
403 if (i == (IntSize - 1) && byte > 127)
404 isNegative = true;
405
406 // Then, Bitwise OR the next byte to the retrieved value by
407 // shifting it into the correct position to continue assembling
408 // the value. Cast the byte to uint64_t because the compiler
409 // might implicity cast to 32-bit int instead.
410 uint64_t shifted = static_cast<uint64_t>(byte) << shiftAmount;
411 retrievedValue |= shifted;
412
413 // Increase the shift amount to select the next byte.
414 shiftAmount += bitsPerByte;
415 }
416
417 return ConvertRetrievedValue(retrievedValue, isNegative);
418 }

422 IntType RetrieveBigEndian() const
423 {
424 bool isNegative = false;
425
426 // Assemble the retrieved value in a 64-bit variable so it can hold
427 // all possible sized values. We make it unsigned so we don't have
428 // acccidental sign changes during the process.
429 uint64_t retrievedValue{ 0 };
430
431 // Because the byte order is reversed for big endian values, we
432 // start the shift amount towards the size and decrease to 0.
433 int shiftAmount = (IntSize - 1) * bitsPerByte;
434
435 for (int i = 0; i < IntSize; ++i)
436 {
437 // First convert the raw byte char to an uint8_t so it can be
438 // bitwised OR'd to the rest of the retrieved value. We keep
439 // everything unsigned to avoid sign conversion errors.
440 uint8_t byte = static_cast<uint8_t>(rawData[i]);
441
442 // Determine if the most significant byte has the sign bit
443 // enabled to determine if the entire number is negative.
444 if (i == 0 && byte > 127)
445 isNegative = true;
446
447 // Then, Bitwise OR the next byte to the retrieved value by
448 // shifting it into the correct position to continue assembling
449 // the value. Cast the byte to uint64_t because the compiler

451 retrievedValue |= static_cast<uint64_t>(byte) << shiftAmount;
452
453 // Decrease the shift amount to select the next byte.
454 shiftAmount -= bitsPerByte;
455 }
456

458 }

463 IntType SignExtend(IntType value) const
464 {
465 int valueTotalBytes = sizeof(value);
466 IntType signExtendedValue = value;
467
468 // Start the loop 1 byte past IntSize as we only need to pad the
469 // extra leading bytes.
470 for (int bytes = IntSize + 1; bytes <= valueTotalBytes; bytes++)
471 {
472 int amountToShift = (bytes - 1) * bitsPerByte;
473 signExtendedValue |= byteMask << amountToShift;
474 }
475
476 return signExtendedValue;
477 }

483 IntType ConvertRetrievedValue(uint64_t retrievedValue, bool isNegative)
484 const
485 {
486 IntType convertedValue{ static_cast<IntType>(retrievedValue) };
487
488 // If we're dealing with a signed, negative number, we may need to
489 // sign extend it by padding it with leading 1's (bytes of 0xFF) to
490 // fill the extra bytes if the size of IntType is > IntSize.
491 if constexpr (std::numeric_limits<IntType>::is_signed)
492 {
493 if (isNegative)
494 {
495 // We only need to pad the most significant bytes above and
496 // beyond IntSize. When IntType matches the size of IntSize,
497 // this code will never run. But in cases where IntType's
498 // size is greater than the size of the raw data, such as
499 // Int24Field, we need to pad the extra bytes.
500 if constexpr (sizeof(IntType) > IntSize)
501 return SignExtend(convertedValue);
502 }
503 }
504
505 return convertedValue;
506 }
